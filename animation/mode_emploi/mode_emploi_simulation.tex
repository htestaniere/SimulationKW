\sectionslide{1}{Prise en main}{Lancer l'outil et réaliser une première simulation}

\begin{frame}{À quoi sert l'outil ?}
  \begin{columns}[T]
  \begin{column}{0.55\textwidth}
    Simuler en boucle fermée un \bou{convoi articulé} ($N$ portions) qui suit une trajectoire, pour :
    \begin{itemize}
      \item comparer des \bou{lois de commande} des articulations (amortissement $c$, vérins hydrauliques) et le \bou{vectoring} de couple ;
      \item mesurer la \bou{qualité du suivi} (tête, queue), les plots touchés, l'amplification arrière ;
      \item tester des \bou{manœuvres} normalisées (ISO 3888-2) ou urbaines, avec urgences ;
      \item analyser des \bou{modes dégradés} (capteurs, vérins, moteurs, adhérence).
    \end{itemize}
  \end{column}
  \begin{column}{0.42\textwidth}
    \begin{block}{Ce que vous obtenez}
      Animation du convoi et de ses efforts, courbes (écart latéral, braquage, couples d'articulation, pressions), indicateurs chiffrés, comparaison avec $c=0$ ou avec le cas nominal.
    \end{block}
    \begin{block}{Ce que ce n'est pas}
      Un modèle de pneu complet ni un outil validé sur essais du convoi : voir les limites (dernière partie).
    \end{block}
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Lancer l'outil}
  \begin{columns}[T]
  \begin{column}{0.52\textwidth}
    \begin{enumerate}
      \item Ouvrir le dossier \texttt{Simulation\_Convoi}.
      \item Double-cliquer sur \texttt{simulation\_convoi.html} : la page s'ouvre dans le navigateur (Chrome, Edge, Firefox, Safari).
      \item Aucune installation, aucun réseau : un seul fichier contient le moteur, l'interface et les logos.
      \item Pour les défauts : \texttt{simulation\_degrade.html}, ou le lien « Modes dégradés » en haut de la page.
    \end{enumerate}
    \medskip
    \begin{alertblock}{Une simulation = un clic}
      Rien ne se calcule tant que l'on n'a pas cliqué sur \textbf{Simuler} : régler les curseurs est sans effet immédiat.
    \end{alertblock}
  \end{column}
  \begin{column}{0.44\textwidth}
    \begin{tabular}{@{}ll@{}}

    \end{tabular}

  \end{columns}
\end{frame}

\begin{frame}{Première simulation en quatre étapes}
  \begin{enumerate}
    \item \bou{Choisir la manœuvre} dans les onglets (ex. \emph{Slalom}).
    \item \bou{Régler} le nombre de portions $N$ (2 à 10) et la vitesse d'avance (5 à 50 km/h).
    \item \bou{Ouvrir si besoin} les menus \emph{Commande pilote}, \emph{Commande des articulations}, \emph{Commande vectoring} (repliés au démarrage).
    \item \bou{Cliquer sur Simuler} : la barre indique l'avancement, puis l'animation démarre dès que le calcul est fini.
  \end{enumerate}
  \medskip
  \begin{block}{Réglage de départ conseillé}
    $N=5$, \SI{20}{km/h}, slalom, $c\approx\SI{2700}{N.m.s/rad}$, dynamique des vérins cochée, vectoring désactivé : le convoi suit la trajectoire (erreur de queue $\approx\SI{45}{cm}$, aucun plot touché, RWA $\approx1{,}1$).
  \end{block}
\end{frame}

\begin{frame}{Menu « Commande pilote »}
  \begin{itemize}
    \item \bou{Gain du pilote sur la direction} $g$ (0,3 à 3, défaut 1) : le conducteur braque l'essieu de tête avec une loi de Stanley ; un gain plus fort corrige plus vite l'écart latéral mais peut osciller.
    \item \bou{Récupération d'énergie au lever de pédale} (0 à 100\,\%, défaut 20\,\%) : fraction de la régénération maximale utilisée quand on lève le pied ; au-delà, c'est le frein \emph{mécanique}.
    \item \bou{Circuit urbain} : accélération latérale de confort (1,5\,m/s$^2$), décélération avant les virages (2,5), accélération en sortie (1,2) : ils fixent le profil de vitesse.
    \item \bou{Urgences} : piéton (distance d'apparition 25\,m, temps de réaction 0,6\,s), évitement (40\,m, décalage 2\,m), décélération d'urgence 7\,m/s$^2$.
  \end{itemize}
\end{frame}

\begin{frame}{Menu « Commande des articulations »}
  \begin{columns}[T]
  \begin{column}{0.55\textwidth}
    \begin{itemize}
      \item \bou{$c$ moyen} (N$\cdot$m$\cdot$s/rad) : curseur à échelle quadratique, $c=10^5(s/1000)^2$ ; 0 = articulations libres.
      \item \bou{Loi de répartition} le long du convoi : uniforme, linéaire (pente), polynomiale, gaussienne ; le graphique montre $c_k$ pour chaque articulation.
      \item Case \bou{optimisé selon $N$ et la vitesse} : applique la table des gains de l'optimisation (N de 3 à 10, 10 à 40 km/h).
      \item Case \bou{adapter $c$ en direct} : $c$ suit la vitesse instantanée du convoi (filtre 0,5\,s) au lieu de la vitesse initiale.
      \item \bou{Dynamique des vérins} : remplace l'amortisseur idéal par le modèle hydraulique ; options « vérins non contrôlés » et « boucle fermée sur l'effort ».
    \end{itemize}
  \end{column}
  \begin{column}{0.42\textwidth}
    \begin{alertblock}{À retenir}
      Modifier $c$ ou la répartition à la main désactive la case « optimisé ».
    \end{alertblock}
    \begin{block}{Ordre de grandeur}
      À \SI{20}{km/h}, $N=5$ : $c\approx\SI{2000}{N.m.s/rad}$, plus fort à l'avant (pente $+0{,}75$).
    \end{block}
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Menu « Commande vectoring »}
  \begin{itemize}
    \item Case \bou{Activer le vectoring} : crée un moment de lacet sur chaque portion par différence de couples entre roues gauche et droite (moteurs-roues).
    \item \bou{Gains} $k_\omega$ et $k_d$ (10\,s$^{-1}$ chacun), \bou{vitesse minimale d'activation} (10\,km/h, montée progressive jusqu'à $1{,}4\times$).
    \item \bou{Couple moteur} : courbes mesurées (traction, régénération à 100\,A ou 50\,A) ou valeur constante (\SI{250}{N.m}).
    \item \bou{Demi-voie} des roues motrices (0,46\,m), \bou{rayon de roue} (0,23\,m).
    \item Le vectoring \bou{n'utilise que la capacité restante} des moteurs et \bou{jamais le frein mécanique} : la régénération passe en premier, la traction de l'autre roue complète.
  \end{itemize}
  \begin{block}{Bon réflexe}
    Comparer deux simulations identiques, avec et sans vectoring : la page affiche aussi la simulation sans vectoring en fantôme.
  \end{block}
\end{frame}

\begin{frame}{Slalom et test de l'élan : plots et contacts}
  \begin{columns}[T]
  \begin{column}{0.52\textwidth}
    \begin{itemize}
      \item \bou{Slalom} : les plots sont sur l'axe, à l'abscisse de chaque maximum d'amplitude ; le convoi les contourne alternativement par la gauche et par la droite.
      \item \bou{ISO 3888-2} : trois couloirs de largeurs $1{,}1b+0{,}25$, $b+1$ et $\max(1{,}3b+0{,}25;3)$, avec $b=\SI{1,5}{m}$.
      \item Chaque portion est un rectangle de largeur \SI{1,5}{m} : un plot touché devient \bou{rouge} et est compté (« plots touchés : n / N »).
      \item À \SI{20}{km/h} le slalom par défaut ne touche aucun plot ; à \SI{30}{km/h} trois sur cinq.
    \end{itemize}
  \end{column}
  \begin{column}{0.44\textwidth}
    \begin{block}{Pour explorer}
      Augmenter la vitesse par pas de 5\,km/h, relever l'erreur de queue et les plots touchés ; changer $N$ et la répartition de $c$ ; activer le vectoring.
    \end{block}
  \end{column}
  \end{columns}
\end{frame}

\sectionslide{4}{Lire et comparer}{Indicateurs, comparaisons, bonnes pratiques de lecture}

\begin{frame}{Les indicateurs}
  \begin{columns}[T]
  \begin{column}{0.54\textwidth}
    \begin{itemize}
      \item \bou{Erreur de la queue} : écart maximal du dernier essieu à la trajectoire (affichée à côté du bouton Simuler).
      \item \bou{Plots touchés} : $n/N$, avec l'instant du premier contact.
      \item \bou{RWA} (amplification arrière, ISO 14791) : rapport des accélérations latérales maximales queue/tête ; $>1$ : la queue est plus sollicitée que la tête.
      \item \bou{Déport} : écart entre la queue et la trajectoire réellement suivie par la tête.
      \item \bou{Couples d'articulation} et leur saturation ; \bou{pressions et courants} des vérins.
      \item \bou{Urgences} : contact oui/non, vitesse d'impact, distance minimale, distance d'arrêt.
    \end{itemize}
  \end{column}
  \begin{column}{0.42\textwidth}
    \begin{block}{Statuts}
      \textbf{fin} : trajectoire terminée.\\
      \textbf{divergence} : le convoi ne suit plus la trajectoire (queue à plus de 12\,m ou angle relatif $>2{,}6$\,rad) : les indicateurs ne sont alors pas significatifs.
    \end{block}
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Comparer deux réglages}
  \begin{enumerate}
    \item \bou{Avec / sans amortissement} : cocher « comparer avec $c=0$ » : la simulation sans amortissement s'affiche en fantôme et ses courbes en trait fin.
    \item \bou{Avec / sans vectoring} : la simulation sans vectoring est calculée et superposée.
    \item \bou{Cas nominal / défaut} (page dégradée) : la simulation sans défaut est la référence.
    \item Relever à chaque fois : erreur de queue, plots, RWA, couples maximaux.
  \end{enumerate}
  \medskip
  \begin{alertblock}{Pour comparer proprement}
    Ne changer qu'un seul paramètre à la fois ; garder la même vitesse, le même $N$ et la même graine (circuit aléatoire, urgences, défauts).
  \end{alertblock}
\end{frame}

\sectionslide{6}{Conseils et limites}{Pour bien utiliser l'outil}

\begin{frame}{Conseils d'utilisation}
  \begin{itemize}
    \item \bou{Partir du réglage par défaut}, puis modifier un paramètre à la fois.
    \item Au-delà de \SI{30}{km/h}, les convois longs divergent dans les manœuvres de test : ce n'est pas un bug, c'est une limite de stabilité (voir le rapport d'optimisation).
    \item Un \bou{calcul long} (urbain, vérins, 10 portions) peut prendre plusieurs secondes ; l'animation part au début dès que le calcul est fini.
    \item \bou{Reproduire un essai} : l'adresse de la page accepte des réglages, p.\,ex. \texttt{simulation\_convoi.html\#slalom,V=20,C=2700} ; onglets : \texttt{volant}, \texttt{elan}, \texttt{slalom}, \texttt{virage}, \texttt{cercle}, \texttt{urbain}, \texttt{brake}, \texttt{dev}.
    \item Noter les \bou{graines} (circuit aléatoire, urgences, défauts) pour pouvoir refaire le même tirage.
    \item Le \bou{rapport complet} (\texttt{rapport\_outil\_simulation.pdf}) donne toutes les équations et les valeurs par défaut.
  \end{itemize}
\end{frame}

\begin{frame}{Limites à garder en tête}
  \begin{itemize}
    \item \bou{Pneus} : loi saturée simple, sans glissement combiné, relaxation ni transfert de charge ; charge verticale $mg/n_P$.
    \item \bou{Vérins} : modèle à deux pressions, sans fuites internes ni température ; la table de vanne sous-estime le débit réel d'un facteur 2 à 3.
    \item \bou{Pilote} : loi de Stanley sans retard de conducteur ; régulation de vitesse proportionnelle idéale.
    \item \bou{Défauts} : modèles simples, notation A à D interne à l'outil.
    \item \bou{Validation} : la dynamique des vérins est confrontée à des essais ; la comparaison du modèle du convoi avec le code MATLAB de référence ou avec un essai réel reste à faire.
  \end{itemize}
  \begin{alertblock}{Conclusion pratique}
    Utiliser l'outil pour \emph{comparer} des réglages et des lois de commande, plutôt que pour prédire une valeur absolue.
  \end{alertblock}
\end{frame}
